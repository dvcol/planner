import Foundation
import PlannerCore

@main struct FixtureGenerator {
  static func main() async throws {
    let directory = URL(fileURLWithPath: CommandLine.arguments[1], isDirectory: true)
    let configuration = PlannerStorageConfiguration(
      storeURL: directory.appendingPathComponent("store/Planner.sqlite"),
      controlURL: directory.appendingPathComponent("control/writer"),
      recoveryDirectoryURL: directory.appendingPathComponent("recovery"),
      processRole: .mainApplication, storageMode: .localOnly)
    let planner = Planner(configuration: configuration)
    guard case .ready(let session) = await planner.bootstrap() else {
      fatalError("Bootstrap failed")
    }
    let operationId = UUID(uuidString: "00000000-0000-4000-8000-000000000941")!
    let content = PlannerItemContentInput(
      title: "Hotel", notes: "Original notes",
      location: PlannerOwnedLocation(
        displayName: "Meeting point", formattedAddress: "Meeting point A",
        coordinate: PlannerCoordinate(latitude: 35, longitude: 139)),
      estimate: PlannerEstimate(minutes: 120, displayUnit: .hour),
      links: [
        PlannerLinkInput(originalUrl: "https://maps.apple.com/?q=Hotel", label: "Map"),
        PlannerLinkInput(originalUrl: "https://example.com/menu", label: "Menu"),
      ])
    guard
      case .applied(let created, .complete(let createdCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: operationId, session: session, command: .createItem(content: content))
      ).outcome, let item = created.generated.first
    else { fatalError("Create failed") }
    guard
      case .applied(_, .complete) = await planner.execute(
        PlannerOperation(
          operationId: UUID(uuidString: "00000000-0000-4000-8000-000000000942")!, session: session,
          command: .setCompletion(scope: .globalItem(itemId: item.id), done: true))
      ).outcome
    else { fatalError("Complete failed") }
    guard
      case .applied(_, .complete(let archivedCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: UUID(uuidString: "00000000-0000-4000-8000-000000000943")!, session: session,
          command: .setArchive(source: item, archived: true))
      ).outcome
    else { fatalError("Archive failed") }
    let form = PlannerScheduleForm.timed(
      start: Date(timeIntervalSinceReferenceDate: 813_200_400.25),
      end: Date(timeIntervalSinceReferenceDate: 813_204_000.75), planningTimeZone: "Asia/Tokyo")
    let scheduleOperationId = UUID(uuidString: "00000000-0000-4000-8000-000000000944")!
    guard
      case .applied(let scheduled, .complete(let checkpoint)) = await planner.execute(
        PlannerOperation(
          operationId: scheduleOperationId, session: session,
          command: .createSchedule(source: item, form: form))
      ).outcome, let schedule = scheduled.generated.first,
      case .source(.schedule(let scheduleRead)) = await planner.read(
        session: session, request: .source(schedule))
    else { fatalError("Schedule failed") }
    guard
      case .applied(_, .complete) = await planner.execute(
        PlannerOperation(
          operationId: UUID(uuidString: "00000000-0000-4000-8000-000000000945")!, session: session,
          command: .removeSchedule(scheduleId: schedule.id))
      ).outcome
    else { fatalError("Remove failed") }
    guard
      case .applied(let replacement, .complete(let finalCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: UUID(uuidString: "00000000-0000-4000-8000-000000000946")!, session: session,
          command: .createSchedule(source: item, form: form))
      ).outcome, let retainedSchedule = replacement.generated.first,
      case .source(.schedule(let retainedRead)) = await planner.read(
        session: session, request: .source(retainedSchedule))
    else { fatalError("Replacement failed") }
    let civilForm = PlannerScheduleForm.allDay(
      start: PlannerCivilDate(year: 2026, month: 10, day: 9),
      end: PlannerCivilDate(year: 2026, month: 10, day: 11))
    guard
      case .applied(let civilResult, .complete(let civilCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: UUID(uuidString: "00000000-0000-4000-8000-000000000947")!, session: session,
          command: .createSchedule(source: item, form: civilForm))
      ).outcome, let civilSchedule = civilResult.generated.first,
      case .source(.schedule(let civilRead)) = await planner.read(
        session: session, request: .source(civilSchedule))
    else { fatalError("Civil creation failed") }
    let listOperationId = UUID(uuidString: "00000000-0000-4000-8000-000000000948")!
    let listContent = PlannerListContentInput(
      name: " Tokyo Food ", notes: "Original List notes",
      color: PlannerColor(red: 0.125, green: 0.5, blue: 0.75, alpha: 1), iconName: "fork.knife")
    guard
      case .applied(let listResult, .complete(let listCreationCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: listOperationId, session: session, command: .createList(content: listContent)
        )
      ).outcome, let list = listResult.generated.first,
      case .source(.list(let beforeList)) = await planner.read(
        session: session, request: .source(list))
    else { fatalError("List creation failed") }
    let listEditOperationId = UUID(uuidString: "00000000-0000-4000-8000-000000000949")!
    guard
      case .applied(_, .complete(let listEditCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: listEditOperationId, session: session,
          command: .editList(
            sourceId: list.id, changes: PlannerListChanges(notes: .set(" Friday booking ")),
            expectedFieldHashes: beforeList.fieldHashes))
      ).outcome
    else { fatalError("List edit failed") }
    let listArchiveOperationId = UUID(uuidString: "00000000-0000-4000-8000-000000000950")!
    guard
      case .applied(_, .complete(let listArchiveCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: listArchiveOperationId, session: session,
          command: .setArchive(source: list, archived: true))
      ).outcome,
      case .source(.list(let listRead)) = await planner.read(
        session: session, request: .source(list))
    else { fatalError("List archive failed") }
    guard
      case .applied(let secondListResult, .complete(let latestCheckpoint)) = await planner.execute(
        PlannerOperation(
          operationId: UUID(uuidString: "00000000-0000-4000-8000-000000000951")!, session: session,
          command: .createList(
            content: PlannerListContentInput(
              name: "Wishlist", color: PlannerColor(red: 0, green: 0, blue: 1, alpha: 1),
              iconName: "")))
      ).outcome, let secondList = secondListResult.generated.first,
      case .source(.list(let secondListRead)) = await planner.read(
        session: session, request: .source(secondList))
    else { fatalError("Second List failed") }
    guard
      case .source(.item(let source)) = await planner.read(session: session, request: .source(item))
    else { fatalError("Read failed") }
    guard
      case .listedNamespaces(let namespaces) = await planner.inspectRecovery(request: .namespaces),
      let namespace = namespaces.first,
      case .selected(let selection) = await planner.inspectRecovery(
        request: .acknowledgedSnapshot(
          namespaceId: namespace.namespaceId, checkpointGeneration: latestCheckpoint))
    else { fatalError("Recovery failed") }
    let backup = try JSONSerialization.jsonObject(with: selection.portableData) as! [String: Any]
    var manifest: [String: Any] = [
      "producerCommit": "984b98f", "datasetId": session.datasetId.uuidString,
      "itemId": item.id.uuidString, "createdAt": source.createdAt.timeIntervalSinceReferenceDate,
      "updatedAt": source.updatedAt.timeIntervalSinceReferenceDate,
      "fieldHashes": Dictionary(
        uniqueKeysWithValues: source.fieldHashes.map { ($0.key.rawValue, $0.value.value) }),
      "creationOperationId": operationId.uuidString, "creationCheckpoint": createdCheckpoint,
      "latestCheckpoint": latestCheckpoint, "archivedCheckpoint": archivedCheckpoint,
      "ownedLinks": backup["ownedLinks"]!, "scheduleId": schedule.id.uuidString,
      "scheduleOperationId": scheduleOperationId.uuidString,
      "scheduleHash": scheduleRead.fieldHashes[.form]!.value, "schedules": backup["schedules"]!,
      "deletionMarkers": backup["deletionMarkers"]!,
      "retainedScheduleId": retainedSchedule.id.uuidString,
      "retainedScheduleHash": retainedRead.fieldHashes[.form]!.value,
      "allDayScheduleId": civilSchedule.id.uuidString,
      "allDayScheduleHash": civilRead.fieldHashes[.form]!.value,
    ]
    manifest["lists"] = (backup["sources"] as! [[String: Any]]).filter {
      $0["kind"] as? String == "list"
    }
    manifest["listMetadata"] = [listRead, secondListRead].map { read in
      [
        "id": read.source.id.uuidString, "createdAt": read.createdAt.timeIntervalSinceReferenceDate,
        "updatedAt": read.updatedAt.timeIntervalSinceReferenceDate,
        "fieldHashes": Dictionary(
          uniqueKeysWithValues: read.fieldHashes.map { ($0.key.rawValue, $0.value.value) }),
      ] as [String: Any]
    }
    manifest["removedScheduleCreationCheckpoint"] = checkpoint
    manifest["retainedScheduleCreationCheckpoint"] = finalCheckpoint
    manifest["listId"] = list.id.uuidString
    manifest["listCreationOperationId"] = listOperationId.uuidString
    manifest["listEditOperationId"] = listEditOperationId.uuidString
    manifest["listArchiveOperationId"] = listArchiveOperationId.uuidString
    manifest["listCreationCheckpoint"] = listCreationCheckpoint
    manifest["listEditCheckpoint"] = listEditCheckpoint
    manifest["listArchiveCheckpoint"] = listArchiveCheckpoint
    manifest["allDayCreationCheckpoint"] = civilCheckpoint
    try JSONSerialization.data(withJSONObject: manifest, options: [.prettyPrinted, .sortedKeys])
      .write(to: directory.appendingPathComponent("manifest.json"))
  }
}
