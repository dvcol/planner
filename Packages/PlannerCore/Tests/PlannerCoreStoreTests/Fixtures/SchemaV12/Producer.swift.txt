import Foundation
import PlannerCore
import Testing

struct Schema12ProducerTests {
  @Test func produceGenuineSchema12Fixture() async throws {
    let directory = URL(fileURLWithPath: "/private/tmp/PlannerSchema12Fixture-5c66eb9")
    let planner = Planner(configuration: .init(
      storeURL: directory.appendingPathComponent("store/Planner.sqlite"),
      controlURL: directory.appendingPathComponent("control/writer"),
      recoveryDirectoryURL: directory.appendingPathComponent("recovery"),
      processRole: .mainApplication, storageMode: .localOnly))
    guard case .ready(let session) = await planner.bootstrap() else { throw CancellationError() }
    func save(_ command: PlannerCommand) async throws -> PlannerAppliedResult {
      guard case .applied(let result, .complete) = await planner.execute(
        .init(operationId: UUID(), session: session, command: command)).outcome else { throw CancellationError() }
      return result
    }
    let item = try #require(try await save(.createItem(content: .init(title: "Hotel"))).generated.first)
    let list = try #require(try await save(.createList(content: .init(name: "Stops"))).generated.first)
    let itinerary = try #require(try await save(.createItinerary(content: .init(title: "Route"))).generated.first)
    let membership = try #require(try await save(.addMembership(itemId: item.id, listId: list.id, placement: .last)).generatedReferences.first)
    var entryIds: [UUID] = []
    for source in [item, item, list, list] {
      let reference = try #require(try await save(.addItineraryEntry(itineraryId: itinerary.id, source: source, placement: .last)).generatedReferences.first)
      guard case .itineraryEntry(let identifier, _, _) = reference else { throw CancellationError() }
      entryIds.append(identifier)
    }
    guard case .membership(let membershipId, _, _) = membership,
      case .source(.itinerary(let initial)) = await planner.read(session: session, request: .source(itinerary)) else { throw CancellationError() }
    _ = try await save(.setCompletion(scope: .appearance(.directItineraryItem(itineraryId: itinerary.id, entryId: entryIds[0])), done: true))
    _ = try await save(.setCompletion(scope: .appearance(.listMembership(listId: list.id, membershipId: membershipId)), done: true))
    _ = try await save(.editItinerary(sourceId: itinerary.id, changes: .init(notes: .set("Updated plan")),
      expectedFieldHashes: [.notes: try #require(initial.fieldHashes[.notes])]))
    _ = try await save(.setArchive(source: list, archived: true))
    guard case .source(.itinerary(let read)) = await planner.read(session: session, request: .source(itinerary)),
      case .listedNamespaces(let namespaces) = await planner.inspectRecovery(request: .namespaces),
      let namespace = namespaces.first,
      case .selected(let recovery) = await planner.inspectRecovery(request: .acknowledgedSnapshot(namespaceId: namespace.namespaceId, checkpointGeneration: 12)) else { throw CancellationError() }
    #expect(namespace.acknowledgedSnapshot?.storageSchemaVersion == 12)
    #expect(read.progress.totalCount == 4)
    #expect(read.progress.doneCount == 1)
    let manifest: [String: Any] = [
      "producerCommit": "5c66eb9", "storageSchemaVersion": 12, "checkpoint": 12,
      "datasetId": session.datasetId.uuidString, "namespaceId": namespace.namespaceId.uuidString,
      "itemId": item.id.uuidString, "listId": list.id.uuidString, "itineraryId": itinerary.id.uuidString,
      "membershipId": membershipId.uuidString, "entryIds": entryIds.map(\.uuidString),
      "itineraryCreatedAt": read.createdAt.timeIntervalSinceReferenceDate,
      "itineraryUpdatedAt": read.updatedAt.timeIntervalSinceReferenceDate,
      "itineraryHashes": Dictionary(uniqueKeysWithValues: read.fieldHashes.map { ($0.key.rawValue, $0.value.value) }),
      "portableDataBase64": recovery.portableData.base64EncodedString()
    ]
    try JSONSerialization.data(withJSONObject: manifest, options: [.prettyPrinted, .sortedKeys]).write(to: directory.appendingPathComponent("manifest.json"))
  }
}
