import Foundation
import PlannerCore

@main struct Schema14Producer {
  static func main() async throws {
    let directory = URL(fileURLWithPath: "/private/tmp/PlannerSchema14Fixture-a47caea")
    let planner = Planner(configuration: .init(
      storeURL: directory.appendingPathComponent("store/Planner.sqlite"),
      controlURL: directory.appendingPathComponent("control/writer"),
      recoveryDirectoryURL: directory.appendingPathComponent("recovery"),
      processRole: .mainApplication, storageMode: .localOnly))
    guard case .ready(let session) = await planner.bootstrap() else { throw CancellationError() }
    func save(_ command: PlannerCommand, operationId: UUID = UUID()) async throws -> PlannerAppliedResult {
      let result = await planner.execute(.init(operationId: operationId, session: session, command: command))
      guard case .applied(let applied, .complete) = result.outcome else {
        throw NSError(domain: String(describing: result.outcome), code: 1)
      }
      return applied
    }
    func generated(_ command: PlannerCommand, operationId: UUID = UUID()) async throws -> PlannerEntityReference {
      guard let reference = try await save(command, operationId: operationId).generated.first else { throw CancellationError() }
      return reference
    }
    let category = try await generated(.createCategory(content: .init(name: "Food")))
    let tag = try await generated(.createTag(content: .init(name: "Rain")))
    let item = try await generated(.createItem(content: .init(title: "Hotel", notes: "Keep",
      estimate: .init(minutes: 30, displayUnit: .minute), links: [.init(originalUrl: "https://example.com/hotel")], categoryIds: [category.id], tagIds: [tag.id])))
    let listCreationIdentifier = UUID()
    let list = try await generated(.createList(content: .init(name: "Stops")), operationId: listCreationIdentifier)
    let itinerary = try await generated(.createItinerary(content: .init(title: "Route", notes: "Keep route",
      links: [.init(originalUrl: "https://example.com/route")], categoryIds: [category.id], tagIds: [tag.id])))
    guard case .membership(let membershipId, _, _) = try await save(
      .addMembership(itemId: item.id, listId: list.id, placement: .last)).generatedReferences.first
    else { throw CancellationError() }
    var entryIds: [UUID] = []
    for source in [item, item, list, list] {
      guard case .itineraryEntry(let identifier, _, _) = try await save(
        .addItineraryEntry(itineraryId: itinerary.id, source: source, placement: .last)).generatedReferences.first
      else { throw CancellationError() }
      entryIds.append(identifier)
    }
    _ = try await save(.setCompletion(scope: .appearance(.directItineraryItem(
      itineraryId: itinerary.id, entryId: entryIds[0])), done: true))
    _ = try await save(.setCompletion(scope: .appearance(.expandedListItem(
      itineraryId: itinerary.id, listEntryId: entryIds[2], membershipId: membershipId)), done: true))
    _ = try await save(.setCompletion(scope: .appearance(.listMembership(
      listId: list.id, membershipId: membershipId)), done: true))
    _ = try await save(.setArchive(source: list, archived: true))
    let timed = try await generated(.createSchedule(source: item, form: .timed(
      start: Date(timeIntervalSinceReferenceDate: 813_200_400), end: nil, planningTimeZone: "Asia/Tokyo")))
    let allDay = try await generated(.createSchedule(source: item, form: .allDay(
      start: .init(year: 2026, month: 10, day: 9), end: .init(year: 2026, month: 10, day: 11))))
    let removed = try await generated(.createSchedule(source: item, form: .timed(
      start: Date(timeIntervalSinceReferenceDate: 813_214_800), end: nil, planningTimeZone: "Asia/Tokyo")))
    _ = try await save(.removeSchedule(scheduleId: removed.id))
    let itinerarySchedule = try await generated(.createSchedule(source: itinerary, form: .allDay(
      start: .init(year: 2026, month: 10, day: 9), end: .init(year: 2026, month: 10, day: 11))))
    guard case .source(.list(let listRead)) = await planner.read(session: session, request: .source(list)),
      case .source(.itinerary(let itineraryRead)) = await planner.read(session: session, request: .source(itinerary)),
      case .source(.item(let itemRead)) = await planner.read(session: session, request: .source(item)),
      case .listedNamespaces(let namespaces) = await planner.inspectRecovery(request: .namespaces),
      let namespace = namespaces.first, let checkpoint = namespace.acknowledgedSnapshot?.checkpointGeneration,
      case .selected(let recovery) = await planner.inspectRecovery(request: .acknowledgedSnapshot(
        namespaceId: namespace.namespaceId, checkpointGeneration: checkpoint))
    else { throw CancellationError() }
    guard namespace.acknowledgedSnapshot?.storageSchemaVersion == 14,
      itineraryRead.progress.totalCount == 4, itineraryRead.progress.doneCount == 2 else { throw CancellationError() }
    let manifest: [String: Any] = [
      "producerCommit": "a47caea2c65d5f8c417c627231a62d79cb277564", "storageSchemaVersion": 14,
      "checkpoint": checkpoint, "datasetId": session.datasetId.uuidString,
      "namespaceId": namespace.namespaceId.uuidString, "itemId": item.id.uuidString,
      "listId": list.id.uuidString, "itineraryId": itinerary.id.uuidString,
      "listCreationIdentifier": listCreationIdentifier.uuidString,
      "listHashes": Dictionary(uniqueKeysWithValues: listRead.fieldHashes.map { ($0.key.rawValue, $0.value.value) }),
      "membershipId": membershipId.uuidString, "entryIds": entryIds.map(\.uuidString),
      "scheduleIds": [timed.id.uuidString, allDay.id.uuidString, itinerarySchedule.id.uuidString],
      "itineraryCreatedAt": itineraryRead.createdAt.timeIntervalSinceReferenceDate,
      "itineraryUpdatedAt": itineraryRead.updatedAt.timeIntervalSinceReferenceDate,
      "itineraryHashes": Dictionary(uniqueKeysWithValues: itineraryRead.fieldHashes.map { ($0.key.rawValue, $0.value.value) }),
      "itemHashes": Dictionary(uniqueKeysWithValues: itemRead.fieldHashes.map { ($0.key.rawValue, $0.value.value) }),
      "portableDataBase64": recovery.portableData.base64EncodedString()
    ]
    try JSONSerialization.data(withJSONObject: manifest, options: [.prettyPrinted, .sortedKeys])
      .write(to: directory.appendingPathComponent("manifest.json"))
  }
}
